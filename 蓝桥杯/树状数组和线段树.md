# 树状数组和线段树
## 树状数组
O(logn) => 单点修改 && 区间查询
![[Screenshot_2026-04-27-19-46-12-90_cb44f87ba7f0946bb27207e460674ef7.png]]
`c[x]` 表示 `( x - lowbit(x), x ]` 的和。
`x` 的父结点是 `x + lowbit(x)` 
#### 求前缀和
```cpp
int res = 0;
for (int i = x; i > 0; i -= lowbit(i)) res += c[i];
return res;
```
#### 更新
```cpp
a[x] += v;
for (int i = x; i <= n; i += lowbit(i)) c[i] += v;
```
### 经典例题
![[IMG_20260428_164819.jpg]]
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100010;

int n, m;
int a[N], c[N];

// lowbit  add  query 要背下来 
int lowbit(int x) {
    return x & -x;
}

void add(int x, int v) {
    for (int i = x; i <= n; i += lowbit(i)) c[i] += v;
}

int query(int x) {
    int res = 0; 
    for (int i = x; i > 0; i -= lowbit(i)) res += c[i];
    return res;
}

int main() {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i ++) scanf("%d", &a[i]);
    for (int i = 1; i <= n; i ++) add(i, a[i]);
    
    while (m --) {
        int k, x, y;
        scanf("%d%d%d", &k, &x, &y);
        if (!k) printf("%d\n", query(y) - query(x - 1));
        else add(x, y);
    }
    
    return 0;
}
```
## 线段树
![[Screenshot_2026-04-30-14-48-18-68_cb44f87ba7f0946bb27207e460674ef7.jpg]]
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100010;
int n, m;
int a[N];
struct Node {
    int l, r;
    int sum;
} tr[N * 4];


void pushup(int u) {
    tr[u].sum = tr[u << 1].sum + tr[u << 1 | 1].sum;
}

void build(int u, int l, int r) {
    if (l == r) {
        tr[u] = {l, r, a[r]};
        return;
    }
    tr[u] = {l, r};
    int mid = l + r >> 1;
    build(u << 1, l, mid);
    build(u << 1 | 1, mid + 1, r);
    pushup(u);
}

int query(int u, int l, int r) {
    if (tr[u].l >= l && tr[u].r <= r) {
        return tr[u].sum;
    }
    int res = 0;
    int mid = tr[u].l + tr[u].r >> 1;
    if (l <= mid) res += query(u << 1, l, r);
    if (r > mid) res += query(u << 1 | 1, l, r);
    return res;
}

// 这个你掌握的不太好
void modify(int u, int x, int v) {
    if (tr[u].l == tr[u].r) {
        tr[u].sum += v;
        return;
    }
    int mid = tr[u].r + tr[u].l >> 1;
    if (x <= mid) modify(u << 1, x, v);
    else modify(u << 1 | 1, x, v);
    pushup(u);
}

int main () {
    scanf("%d%d", &n, &m);
    for (int i = 1; i <= n; i ++) scanf("%d", &a[i]);
    build(1, 1, n);
    while (m --) {
        int k, x, y;
        scanf("%d%d%d", &k, &x, &y);
        if (k == 0) printf("%d\n", query(1, x, y));
        else modify(1, x, y);
    }
    
    return 0;
}
```
